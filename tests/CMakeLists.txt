# The family's dependency rule and its enforcement (PLAN.md
# 4.2): async, bridge and rational each depend on tap::dsp only, and a
# sibling appears only under an engine's tests/ and examples/. Four checks
# per engine, each registered under the engine's label so every job that
# runs that engine's tests runs them, plus D13's version-macro check under
# every label. Nothing here links an engine to a sibling. The compile checks are built BY their
# test (cmake --build of an EXCLUDE_FROM_ALL target), so a leaked include is
# a failing test, not a broken build.
set(_family ${CMAKE_CURRENT_SOURCE_DIR}/family)
set(_header_count_async 6)
set(_header_count_bridge 5)
# rational grows to six by its plan's layout (rational/PLAN.md section 4);
# re-pinned by the milestone that adds a header (M1: rational.h, ratio.h,
# design.h; M3: stage.h, converter.h; M4 adds chain.h).
set(_header_count_rational 6)

set(_engines async bridge rational)
foreach(_engine IN LISTS _engines)
    set(_other ${_engines})
    list(REMOVE_ITEM _other ${_engine})
    set(_include ${CMAKE_SOURCE_DIR}/${_engine}/include)

    # 1. Link interface: exactly tap::dsp. Read at configure time, asserted
    #    at test time, so the check is a test rather than a configure step.
    get_target_property(_links tap_sr_${_engine} INTERFACE_LINK_LIBRARIES)
    add_test(NAME ${_engine}.Family.LinkInterfaceIsDspOnly
             COMMAND ${CMAKE_COMMAND} -DENGINE=${_engine} "-DLINKS=${_links}" -DEXPECT=tap::dsp
                     -P ${_family}/check_link_interface.cmake)

    # 2. Header isolation: every public header in its own TU, with only
    #    tap::dsp and this engine's include directory on the path.
    file(GLOB_RECURSE _headers RELATIVE ${_include} CONFIGURE_DEPENDS ${_include}/*.h)
    set(_tus)
    foreach(_h IN LISTS _headers)
        string(MAKE_C_IDENTIFIER ${_h} _id)
        set(_tu ${CMAKE_CURRENT_BINARY_DIR}/isolation/${_engine}_${_id}.cpp)
        file(WRITE ${_tu} "// Generated by tests/CMakeLists.txt: 4.2 check 2.\n#include \"${_h}\"\n")
        list(APPEND _tus ${_tu})
    endforeach()
    add_library(tap_sr_${_engine}_header_isolation STATIC EXCLUDE_FROM_ALL ${_tus})
    target_include_directories(tap_sr_${_engine}_header_isolation PRIVATE ${_include})
    target_link_libraries(tap_sr_${_engine}_header_isolation PRIVATE tap::dsp)
    target_compile_features(tap_sr_${_engine}_header_isolation PRIVATE cxx_std_20)
    add_test(NAME ${_engine}.Family.HeadersCompileInIsolation
             COMMAND ${CMAKE_COMMAND} --build ${CMAKE_BINARY_DIR} --config $<CONFIG>
                     --target tap_sr_${_engine}_header_isolation)

    # 3. Include grep: the public headers name no srt/, no sibling
    #    tap/sr/<other>/ (either sibling), no ../ and no sibling __has_include.
    add_test(NAME ${_engine}.Family.IncludesStayInEngine
             COMMAND ${CMAKE_COMMAND} -DENGINE=${_engine} -DINCLUDE=${_include} "-DOTHER=${_other}"
                     -P ${_family}/check_includes.cmake)

    # 4. Header count, pinned: a header cannot appear or vanish unnoticed.
    add_test(NAME ${_engine}.Family.HeaderCountIsPinned
             COMMAND ${CMAKE_COMMAND} -DENGINE=${_engine} -DINCLUDE=${_include}
                     -DEXPECT=${_header_count_${_engine}} -P ${_family}/check_header_count.cmake)

    set_tests_properties(${_engine}.Family.LinkInterfaceIsDspOnly
                         ${_engine}.Family.HeadersCompileInIsolation
                         ${_engine}.Family.IncludesStayInEngine
                         ${_engine}.Family.HeaderCountIsPinned
                         PROPERTIES LABELS ${_engine})
endforeach()

# 5. The C ABI's exported symbol set (when the libraries are built): nm on
#    the shared library against tests/family/symbols_<engine>.txt, so a
#    leaked C++ symbol or a stray entry point is a failing test (the 2026-10
#    audit, F04). An nm test, so the ELF and Mach-O hosts only; Windows
#    gates the DLLs through the CApi suites.
if(TAP_SR_BUILD_CAPI AND NOT WIN32 AND NOT CMAKE_CROSSCOMPILING)
    find_program(TAP_SR_NM NAMES nm llvm-nm)
    if(TAP_SR_NM)
        foreach(_engine IN LISTS _engines)
            add_test(NAME ${_engine}.Family.ExportedSymbolsArePinned
                     COMMAND ${CMAKE_COMMAND} -DENGINE=${_engine}
                             -DLIBRARY=$<TARGET_FILE:tap_sr_${_engine}_capi>
                             -DEXPECTED=${_family}/symbols_${_engine}.txt
                             -DNM=${TAP_SR_NM}
                             -P ${_family}/exported_symbols.cmake)
            set_tests_properties(${_engine}.Family.ExportedSymbolsArePinned PROPERTIES LABELS ${_engine})
        endforeach()
    else()
        message(WARNING "nm not found: the exported-symbol tests are not registered")
    endif()
endif()

# 6. The READMEs' quick starts compile: the first ```cpp block of the root
#    README and of each engine's, extracted at configure time into a TU
#    (tests/family/readme_snippet.cmake, a prelude declaring the caller's
#    buffers), built by its test. A quick start that stops compiling is a
#    red check, not an audit finding (2026-10, S5).
set(_readme_dir ${CMAKE_BINARY_DIR}/tests/readme)
file(MAKE_DIRECTORY ${_readme_dir})
foreach(_name IN ITEMS family async bridge rational)
    if(_name STREQUAL family)
        set(_readme ${CMAKE_SOURCE_DIR}/README.md)
        set(_labels "async;bridge;rational")
    else()
        set(_readme ${CMAKE_SOURCE_DIR}/${_name}/README.md)
        set(_labels ${_name})
    endif()
    set(_prelude ${_family}/readme/${_name}_prelude.h)
    set(_tu ${_readme_dir}/${_name}_quick_start.cpp)
    execute_process(COMMAND ${CMAKE_COMMAND} -DREADME=${_readme} -DPRELUDE=${_prelude} -DOUT=${_tu}
                            -P ${_family}/readme_snippet.cmake
                    RESULT_VARIABLE _rc OUTPUT_QUIET)
    if(NOT _rc EQUAL 0)
        message(FATAL_ERROR "could not extract the quick start of ${_readme}")
    endif()
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${_readme})
    add_executable(tap_sr_readme_${_name} EXCLUDE_FROM_ALL ${_tu})
    target_include_directories(tap_sr_readme_${_name} PRIVATE
        ${CMAKE_SOURCE_DIR}/async/include ${CMAKE_SOURCE_DIR}/bridge/include ${CMAKE_SOURCE_DIR}/rational/include)
    target_link_libraries(tap_sr_readme_${_name} PRIVATE tap::dsp)
    target_compile_features(tap_sr_readme_${_name} PRIVATE cxx_std_20)
    if(Threads_FOUND)
        target_link_libraries(tap_sr_readme_${_name} PRIVATE Threads::Threads)
    endif()
    add_test(NAME ${_name}.Family.ReadmeQuickStartCompiles
             COMMAND ${CMAKE_COMMAND} --build ${CMAKE_BINARY_DIR} --config $<CONFIG>
                     --target tap_sr_readme_${_name})
    set_tests_properties(${_name}.Family.ReadmeQuickStartCompiles PROPERTIES LABELS "${_labels}")
endforeach()

# D13: TAP_SR_VERSION_* is defined in each of the three umbrella headers (no
# shared header, so check 1 holds) and must agree. Compile-only, built by its test, so it
# runs on every leg including bare metal.
add_library(tap_sr_family_version STATIC EXCLUDE_FROM_ALL ${_family}/version_macros.cpp)
target_include_directories(tap_sr_family_version PRIVATE
    ${CMAKE_SOURCE_DIR}/async/include ${CMAKE_SOURCE_DIR}/bridge/include
    ${CMAKE_SOURCE_DIR}/rational/include)
target_link_libraries(tap_sr_family_version PRIVATE tap::dsp)
target_compile_features(tap_sr_family_version PRIVATE cxx_std_20)
add_test(NAME family.VersionMacrosAgree
         COMMAND ${CMAKE_COMMAND} --build ${CMAKE_BINARY_DIR} --config $<CONFIG>
                 --target tap_sr_family_version)
set_tests_properties(family.VersionMacrosAgree PROPERTIES LABELS "async;bridge;rational")
